cmake_minimum_required(VERSION 3.22)
project(minime_rime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "../../../.." ABSOLUTE)
set(SOURCES "${ROOT}/.tools/native-sources")
set(RIME "${SOURCES}/librime-1.16.1")
set(OPENCC "${SOURCES}/OpenCC-556ed22496d650bd0b13b6c163be9814637970ae")
set(MARISA "${SOURCES}/marisa-trie-3e87d53b78e15f2f43783d5e376561a8c9722051")
set(YAML "${SOURCES}/yaml-cpp-2f86d13775d119edbb69af52e5f566fd65c6953b")
set(LEVELDB "${SOURCES}/leveldb-99b3c03b3284f5886f9ef9a4ef703d57373e61be")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
set(ENABLE_TIMESTAMP OFF CACHE BOOL "" FORCE)
set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${YAML}" yaml)
add_subdirectory("${LEVELDB}" leveldb)
add_subdirectory("${MARISA}" marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
set(LIBMARISA marisa)
set(ENABLE_DARTS OFF CACHE BOOL "" FORCE)
include_directories("${MARISA}/include")
add_subdirectory("${OPENCC}" opencc)
# librime's Find modules accept already-created dependency targets.
set(YamlCpp_LIBRARY yaml-cpp)
set(YamlCpp_INCLUDE_PATH "${YAML}/include")
set(YamlCpp_NEW_API "${YAML}/include")
set(LevelDb_LIBRARY leveldb)
set(LevelDb_INCLUDE_PATH "${LEVELDB}/include")
set(Marisa_LIBRARY marisa)
set(Marisa_INCLUDE_PATH "${MARISA}/include")
set(Opencc_LIBRARY libopencc)
file(GLOB OPENCC_HEADERS "${OPENCC}/src/*.h" "${OPENCC}/src/*.hpp")
file(COPY ${OPENCC_HEADERS} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencc")
file(COPY "${CMAKE_CURRENT_BINARY_DIR}/opencc/src/opencc_config.h" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencc")
set(Opencc_INCLUDE_PATH "${CMAKE_CURRENT_BINARY_DIR}/include")
set(BOOST_ROOT "${SOURCES}/boost_1_87_0")
set(Boost_INCLUDE_DIR "${BOOST_ROOT}" CACHE PATH "" FORCE)
set(Boost_NO_BOOST_CMAKE ON)
set(X11Keysym "${RIME}/include")
add_subdirectory("${RIME}" rime)
target_compile_options(rime-static PRIVATE -Wno-error=deprecated-declarations)
add_library(minime_rime SHARED rime_jni.cpp)
target_include_directories(minime_rime PRIVATE "${RIME}/src" "${CMAKE_CURRENT_BINARY_DIR}/rime/src" "${BOOST_ROOT}")
target_link_libraries(minime_rime PRIVATE rime-static)
target_link_options(minime_rime PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  # Boost's embedded GDB printer uses @progbits, which ARM treats as a comment.
  target_compile_definitions(rime-static PRIVATE BOOST_ALL_NO_EMBEDDED_GDB_SCRIPTS)
  target_compile_definitions(minime_rime PRIVATE BOOST_ALL_NO_EMBEDDED_GDB_SCRIPTS)
endif()
