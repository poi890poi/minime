cmake_minimum_required(VERSION 3.22)
project(minime_desktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
get_filename_component(ROOT "../.." ABSOLUTE)
set(RIME "${ROOT}/.tools/native-sources/librime-1.16.1")
set(SOURCES "${ROOT}/.tools/native-sources")
set(YAML "${SOURCES}/yaml-cpp-2f86d13775d119edbb69af52e5f566fd65c6953b")
set(LEVELDB "${SOURCES}/leveldb-99b3c03b3284f5886f9ef9a4ef703d57373e61be")
set(MARISA "${SOURCES}/marisa-trie-3e87d53b78e15f2f43783d5e376561a8c9722051")
set(OPENCC "${SOURCES}/OpenCC-556ed22496d650bd0b13b6c163be9814637970ae")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_MSVC_SHARED_RT OFF CACHE BOOL "" FORCE)
add_subdirectory("${YAML}" yaml)
add_subdirectory("${LEVELDB}" leveldb)
add_subdirectory("${MARISA}" marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
set(LIBMARISA marisa)
set(ENABLE_DARTS OFF CACHE BOOL "" FORCE)
include_directories("${MARISA}/include")
add_subdirectory("${OPENCC}" opencc)
target_compile_options(libopencc PRIVATE /clang:-Wno-c++98-compat /clang:-Wno-c++98-compat-pedantic)
file(GLOB opencc_headers "${OPENCC}/src/*.h" "${OPENCC}/src/*.hpp")
file(COPY ${opencc_headers} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencc")
file(COPY "${CMAKE_CURRENT_BINARY_DIR}/opencc/src/opencc_config.h" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencc")
file(GLOB source "${RIME}/src/*.cc" "${RIME}/src/rime/*.cc"
  "${RIME}/src/rime/algo/*.cc" "${RIME}/src/rime/config/*.cc"
  "${RIME}/src/rime/dict/*.cc" "${RIME}/src/rime/gear/*.cc" "${RIME}/src/rime/lever/*.cc")
configure_file("${RIME}/src/rime/build_config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/include/rime/build_config.h")
add_library(minime_rime STATIC ${source})
set_source_files_properties("${RIME}/src/rime/setup.cc" PROPERTIES COMPILE_DEFINITIONS "RIME_EXTRA_MODULES=")
target_include_directories(minime_rime PUBLIC "${RIME}/src" "${RIME}/include"
  "${CMAKE_CURRENT_BINARY_DIR}/include" "${ROOT}/.tools/native-sources/boost_1_87_0"
  "${YAML}/include" "${LEVELDB}/include")
target_compile_definitions(minime_rime PUBLIC YAML_CPP_STATIC_DEFINE Opencc_BUILT_AS_STATIC NOMINMAX RIME_NO_TIMESTAMP RIME_VERSION="1.16.1")
target_compile_options(minime_rime PUBLIC /utf-8 /wd4244 /wd4996)
target_link_libraries(minime_rime PUBLIC yaml-cpp leveldb marisa libopencc shlwapi)
option(MINIME_SOURCE_ONLY "Experimental native construction ablation" ON)
if(MINIME_SOURCE_ONLY)
  include("${ROOT}/tools/native-source-only.cmake")
  minime_source_only(minime_rime)
endif()
add_executable(probe "${ROOT}/tools/desktop_rime_metadata.cpp")
target_compile_definitions(probe PRIVATE MINIME_STATIC_RIME)
target_link_libraries(probe PRIVATE minime_rime)
