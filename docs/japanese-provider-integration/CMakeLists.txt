cmake_minimum_required(VERSION 3.22)
project(minime_japanese_evaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ENABLE_ZENZ OFF CACHE BOOL "" FORCE)
set(BUILD_MOZC_FETCH OFF CACHE BOOL "" FORCE)
add_subdirectory("${UPSTREAM}" upstream)
if(ANDROID)
add_library(minime_japanese_test SHARED Native.cpp)
target_include_directories(minime_japanese_test PRIVATE "${UPSTREAM}" "${UPSTREAM}/src" "${UPSTREAM}/src/dictionary_builder" "${UPSTREAM}/src/dictionary_builder/louds_builder")
target_link_libraries(minime_japanese_test PRIVATE path_algorithm graph_builder token_array louds_utf16 connection_id)
target_link_options(minime_japanese_test PRIVATE "-Wl,-z,max-page-size=16384")
else()
add_executable(converter-server-portable ../japanese-engine-benchmark/ConverterServer.cpp)
target_include_directories(converter-server-portable PRIVATE "${UPSTREAM}" "${UPSTREAM}/src" "${UPSTREAM}/src/dictionary_builder" "${UPSTREAM}/src/dictionary_builder/louds_builder")
target_link_libraries(converter-server-portable PRIVATE path_algorithm graph_builder token_array louds_utf16 connection_id)
add_executable(beam-regression BeamRegression.cpp)
target_link_libraries(beam-regression PRIVATE path_algorithm graph_builder token_array louds_utf16 connection_id)
endif()
